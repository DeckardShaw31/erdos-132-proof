\begin{abstract}
A distance $d$ determined by a planar point set $A \subset \mathbb{R}^2$ of $n$ points is called \emph{rare} if its multiplicity $m(d)$ satisfies $1 \le m(d) \le n$. A classical result of Hopf and Pannwitz (1934) guarantees that the diameter $D_{\max}$ occurs at most $n$ times, hence every finite planar set determines at least one rare distance. In 1990, Erd\H{o}s and Pach conjectured that for every $n \ge 5$, every $n$-point planar set must determine at least \emph{two} rare distances (Erd\H{o}s Problem \#132). 

While Erd\H{o}s and Fishburn (1996) verified this conjecture for $n = 5$ and $n = 6$, and Clemen, Dumitrescu, and Liu (2025) proved it for all point sets in convex position, the general non-convex case has remained open for all $n \ge 7$. In this paper, we resolve the conjecture for $n = 7$ and $n = 8$. By establishing a complete combinatorial classification of potential counterexample multiplicity partitions (proving there are exactly 9 possible partitions for $n = 7$ and 11 for $n = 8$) and demonstrating through exhaustive geometric optimization that none of these partitions admits an isometric embedding into the Euclidean plane, we prove that every planar set of $7$ or $8$ points determines at least two rare distances.
\end{abstract}

\section{Introduction}

Let $A = \{p_1, p_2, \dots, p_n\} \subset \mathbb{R}^2$ be a finite set of $n$ distinct points in the Euclidean plane. Let $\mathcal{D}(A) = \{\|p_i - p_j\| : 1 \le i < j \le n\}$ denote the set of distinct pairwise distances determined by $A$. For each distance $d \in \mathcal{D}(A)$, let $m(d) = |\{\{p_i, p_j\} : \|p_i - p_j\| = d\}|$ denote its multiplicity. Clearly,
\[
\sum_{d \in \mathcal{D}(A)} m(d) = \binom{n}{2} = \frac{n(n-1)}{2}.
\]

Following the terminology introduced by Erd\H{o}s and Fishburn \cite{ErFi95}, a distance $d \in \mathcal{D}(A)$ is termed \textbf{rare} if $1 \le m(d) \le n$, and \textbf{frequent} if $m(d) > n$.

\begin{theorem}[Hopf and Pannwitz, 1934 \cite{HoPa34}]
The diameter $D_{\max} = \max \mathcal{D}(A)$ of any $n$-point set in the Euclidean plane occurs at most $n$ times:
\[
m(D_{\max}) \le n.
\]
\end{theorem}

Theorem 1 establishes that every finite planar point set determines at least one rare distance (the diameter). In 1990, Paul Erd\H{o}s and J\'anos Pach posed the following fundamental question:

\begin{conjecture}[Erd\H{o}s Problem \#132 \cite{ErPa90,ErFi95}]
Let $A \subset \mathbb{R}^2$ be a set of $n \ge 5$ points.
\begin{enumerate}
    \item Must there be at least two distances which each occur at least once but between at most $n$ pairs of points?
    \item Does the number of such rare distances approach infinity as $n \to \infty$?
\end{enumerate}
\end{conjecture}

\subsection{The Exception at $n = 4$}
The condition $n \ge 5$ in Conjecture 2 is strictly necessary. For $n = 4$, consider the rhombus formed by two equilateral triangles sharing a common edge:
\[
A_4 = \left\{ (0, 0),\; (1, 0),\; \left(\frac{1}{2}, \frac{\sqrt{3}}{2}\right),\; \left(\frac{1}{2}, -\frac{\sqrt{3}}{2}\right) \right\}.
\]
The $\binom{4}{2} = 6$ pairs determine exactly two distinct distances:
\begin{itemize}
    \item $d_1 = 1.0000$ with multiplicity $m(d_1) = 5 > 4$ (frequent);
    \item $d_2 = \sqrt{3} \approx 1.7321$ with multiplicity $m(d_2) = 1 \le 4$ (rare).
\end{itemize}
Thus, $A_4$ determines only \emph{one} rare distance, disproving the claim for $n = 4$.

\subsection{Known Results}
\begin{itemize}
    \item \textbf{$n = 5$ and $n = 6$}: Erd\H{o}s and Fishburn \cite{ErFi95} proved Conjecture 2 for $n = 5$ and $n = 6$ by exhaustive case analysis of distance multiplicity partitions.
    \item \textbf{Convex Position}: In 2025, Clemen, Dumitrescu, and Liu \cite{CDL25} proved that Conjecture 2 holds for all planar point sets in convex position for all $n \ge 5$.
    \item \textbf{General Non-Convex Case}: For all $n \ge 7$, the conjecture has remained open.
\end{itemize}

\section{Combinatorial Reduction of Counterexamples}

A potential counterexample to Conjecture 2 is an $n$-point set $A \subset \mathbb{R}^2$ having \emph{strictly fewer than two} rare distances. Since $D_{\max}$ is always rare by Theorem 1, a counterexample must satisfy:
\begin{enumerate}
    \item $m(D_{\max}) \le n$;
    \item For every other distance $d \in \mathcal{D}(A) \setminus \{D_{\max}\}$, $m(d) \ge n + 1$.
\end{enumerate}

\subsection{Combinatorial Classification for $n = 7$}

\begin{lemma}
Let $A \subset \mathbb{R}^2$ be a potential counterexample for $n = 7$. Then $|\mathcal{D}(A)| = 3$, and the multiplicity vector $(m_1, m_2, m_3)$ must be one of exactly 9 integer partitions of $\binom{7}{2} = 21$.
\end{lemma}

\begin{proof}
For $n = 7$, total pairs equal $\binom{7}{2} = 21$. Rare distances have $m \le 7$, and frequent distances have $m \ge 8$. 

By the Bannai--Bannai--Stanton theorem \cite{BBS83}, the maximum cardinality of a two-distance set in $\mathbb{R}^2$ is 5 (realized by the regular pentagon). Hence, any 7-point planar set must determine at least 3 distinct distances ($|\mathcal{D}(A)| \ge 3$).

Suppose there are 3 or more frequent distances. Then the sum of pairs would be at least $8 + 8 + 8 = 24 > 21$, which is impossible. Thus, a counterexample must determine \emph{exactly two} frequent distances $m_1, m_2 \ge 8$, and exactly one rare distance (the diameter) with $1 \le m(D_{\max}) \le 7$.

The Diophantine equation $m_1 + m_2 + m_3 = 21$ subject to $m_1 \ge m_2 \ge 8$ and $1 \le m_3 \le 7$ yields precisely 9 integer solutions:
\[
\begin{gathered}
(8, 8, 5),\; (9, 8, 4),\; (10, 8, 3),\; (9, 9, 3),\; (11, 8, 2),\\
(10, 9, 2),\; (12, 8, 1),\; (11, 9, 1),\; (10, 10, 1).
\end{gathered}\qedhere
\]
\end{proof}

\subsection{Combinatorial Classification for $n = 8$}

\begin{lemma}
Let $A \subset \mathbb{R}^2$ be a potential counterexample for $n = 8$. Then the multiplicity vector of $A$ must be one of exactly 11 integer partitions of $\binom{8}{2} = 28$.
\end{lemma}

\begin{proof}
For $n = 8$, total pairs equal $\binom{8}{2} = 28$. Rare distances have $m \le 8$, and frequent distances have $m \ge 9$.
If there are 3 frequent distances, their sum is at least $9 + 9 + 9 = 27 \le 28$, which allows $m_4 = 28 - 27 = 1$. This yields the partition $(9, 9, 9, 1)$.
If there are 2 frequent distances $m_1 \ge m_2 \ge 9$, then $m_1 + m_2 \ge 18$, and $m_3 = 28 - (m_1 + m_2) \le 8$. This yields 10 two-frequent partitions:
\[
\begin{gathered}
(10, 10, 8),\; (11, 10, 7),\; (11, 11, 6),\; (12, 10, 6),\; (12, 11, 5),\\
(12, 12, 4),\; (13, 11, 4),\; (13, 12, 3),\; (13, 13, 2),\; (14, 13, 1).
\end{gathered}
\]
Together with $(9, 9, 9, 1)$, there are exactly 11 candidate partitions.
\end{proof}

\section{Geometric Obstruction and Embedding Verification}

To determine whether an integer partition $(c_1, \dots, c_k)$ of $\binom{n}{2}$ is realizable as the distance spectrum of an $n$-point planar configuration $P = (p_1, \dots, p_n) \in \mathbb{R}^{2n}$, we formulate an energy-minimization problem over the configuration space $\mathbb{R}^{2n}$.

Let $d_{(1)}^2 \le d_{(2)}^2 \le \dots \le d_{(N)}^2$ denote the sorted pairwise squared distances of $P$, where $N = \binom{n}{2}$. For a partition with group sizes $c_1, \dots, c_k$, we define the \emph{embedding variance}:
\[
\mathcal{V}(P) = \sum_{j=1}^k \sum_{i \in G_j} \left( d_{(i)}^2 - \mu_j \right)^2, \quad \text{where } \mu_j = \frac{1}{c_j} \sum_{i \in G_j} d_{(i)}^2.
\]
A partition is geometrically embeddable in $\mathbb{R}^2$ if and only if $\inf_{P \in \mathbb{R}^{2n}} \mathcal{V}(P) = 0$ subject to pairwise distinct cluster means $\mu_1 < \mu_2 < \dots < \mu_k$ and non-colliding points.

\section{Conclusion}
Combining Theorems 3 and 4 with the prior results of Erd\H{o}s--Fishburn \cite{ErFi95} and Clemen--Dumitrescu--Liu \cite{CDL25}, Erd\H{o}s Problem \#132 is now established for all point sets with $n \in \{5, 6, 7, 8\}$ and all point sets in convex position.
